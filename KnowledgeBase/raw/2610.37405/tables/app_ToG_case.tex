 `Kenya -> named after -> Mount Kenya`;
 `diplomat -> named after -> Diplomatstaden`;

\textbf{Turn 3:}

 `Mount Kenya -> country -> Kenya`;
 `Embraer -> headquarters location -> São José dos Campos`;
 `Category:Dutch politicians -> is a list of -> human`;

\textbf{Turn 4:}

 `Kenya -> named after -> Mount Kenya`;
